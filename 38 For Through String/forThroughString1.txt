FOR THROUGH STRING

Example Script 
---------------------------------------------------------------------
Example 1

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal
	
	:: A B C D E F G H I J

	for /f %%a in ( "This is a string" ) do (
		echo %%a %%b %%c %%d		
	)
	

endlocal
goto :eof
----------------------------------------------------------------------

---------------------------------------------------------------------
Example 2

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal
	
	:: A B C D E F G H I J	

	for /f "token=1-9" %%a in ( "This is a string" ) do (
		echo %%a %%b %%c %%d		
	)
	

endlocal
goto :eof
----------------------------------------------------------------------

Example 3

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal
	
	:: A B C D E F G H I J	
	
	set string=This is a string
	
	for /f "token=1-9" %%g in ( "!string!" ) do (
		echo %%a %%b %%c %%d		
	)
	

endlocal
goto :eof
