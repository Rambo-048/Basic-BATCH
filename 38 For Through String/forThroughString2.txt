FOR THROUGH STRING

Example Script 
---------------------------------------------------------------------
Example 1

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal
	
	:: G H I J K L M N O P	

	for /f %%g in ( "Sometimes We Will Never Know The Value Of A Moment Until It Becomes Memory" ) do (
		echo %%g %%h %%i %%j %%k		
	)
	

endlocal
goto :eof

setlocal

Example 3

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal
	
	:: G H I J K L M N O P		
	
	set string=Things End, People Change. And You Know What? Life Goes On.
	
	for /f "token=1-9" %%g in ( "!string!" ) do (
		echo %%g %%j %%k %%l %%m %%n	
	)
	

endlocal
