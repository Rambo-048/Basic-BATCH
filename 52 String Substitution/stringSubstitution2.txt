STRING SUBSTITUTION :

EXAMPLE SCRIPT :
---------------------------------------
Example 1

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal
	set string=The Only Way To Do Great Work Is To Love What You Do

	echo !string:Great=Good!

endlocal
goto :eof
------------------------------------------------------------------------

------------------------------------------------------------------------
Example 2

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal

	set string=THANKFUL, GREATFUL, BLESSED
	set string=!string:THANKFUL=BE BRAVE!
	set string=!string:GREATFUL=BE STRONG!

	echo !string:BLESSED=BE KIND!

endlocal
goto :eof
