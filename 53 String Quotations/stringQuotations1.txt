STRING QUOTATIONS :

EXAMPLE SCRIPT
--------------------------------------
Example 1 (With Quotations Mark)

@echo off
setlocal EnableDelayedExpansion
goto :main

:main
setlocal

	set string="When You Focus On The Good, The Good Gets Better"

	echo !string!

endlocal
goto :eof
--------------------------------------

---------------------------------------------------------------------------------
Example 2 (No Quotations Mark --> Use for)

@echo off
setlocal EnableDelayedExpansion
goto :main

:main
setlocal

	set string="When You Focus On The Good, The Good Gets Better"
	for /f "useback tokens=*" %%g in ('!string!') do set string=%%~g

	echo !string!

endlocal
goto :eof
---------------------------------------------------------------------------------
