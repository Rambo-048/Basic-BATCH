CREATE STRING FUNCTIONS

EXAMPLE SCRIPT
------------------------------------------------------------
Example 1

@echo off

set string=%2
set /a VALUE=20

for /f "useback tokens=*" %%a in ('%string%') do (
	
	if %string% equ %%~a (
		set /a VALUE=10
	)	
	
	set string %%a
)

echo %2 > %TEMP%\tempfile.txt

for %%g in (%TEMP%\tempfile.txt) do (
	set /a %1_length=%%~zg - %VALUE%
)

del %TEMP%\tempfile.txt
set %1=%string%

Example 2

@echo off

set string=%2


for /f "useback tokens=*" %%a in ('%string%') do (
	
	set string=%%~a
)

echo %2 > %TEMP%\tempfile.txt

for %%g in (%TEMP%\tempfile.txt) do (
	set /a %1_length=%%~zg - 5
)

del %TEMP%\tempfile.txt
set %1=%string%
