CREATE VARIABLE EXPANSION WITHOUT SETLOCAL :

- run => regedit
- reg add /?
- open cmd as adminitrator

EXAMPLE SCRIPT :
--------------------------------------------------------------------------------------------
Example 1 (ENABLE DELAYED EXPANSION)

@echo off
reg add "HKLM\Software\Microsoft\Command Processor" /v "DelayedExpansion" /t REG_DWORD /d 1 /f

-------------------------------------------------------------------------------------------

--------------------------------------------------------------------------------------------
Example 2 (ENABLE DELAYED EXPANSION)

@echo off
reg add "HKLM\Software\Microsoft\Command Processor" /v "VariableExpansion" /t REG_DWORD /d 1 /f > nul

-------------------------------------------------------------------------------------------

--------------------------------------------------------------------------------------------
Example 3 (DISABLE DELAYED EXPANSION)

@echo off

reg add "HKLM\Software\Microsoft\Command Processor" /v "DelayedExpansion" /t REG_DWORD /d 0 /f

-------------------------------------------------------------------------------------------
