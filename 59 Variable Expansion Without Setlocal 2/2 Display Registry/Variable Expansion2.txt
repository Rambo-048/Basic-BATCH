VARIABLE EXPANSION WITHOUT SETLOCAL :

EXAMPLE SCRIPT :
-------------------------------------------------------------------------------------------------------
Example 1

@echo off

reg query "HKCU\Console" /v "ColorTable10"
--------------------------------------------------------------------------------------------------------

---------------------------------------------------------------------------------------------------------
Example 2

@echo off

for /f %%g in ('reg query "HKCU\Console" /v "ColorTable10"') do (
	echo %%g
)
-----------------------------------------------------------------------------------------------------------

---------------------------------------------------------------------------------------------------------
Example 3

@echo off

for /f "tokens=*" %%g in ('HKCU\Console" /v "ColorTable10"') do (
	echo %%g
)
-----------------------------------------------------------------------------------------------------------

---------------------------------------------------------------------------------------------------------
Example 4

@echo off

for /f "tokens=3" %%g in ('reg query "HKCU\Console" /v "ColorTable10"') do (
	echo %%g
)
-----------------------------------------------------------------------------------------------------------
