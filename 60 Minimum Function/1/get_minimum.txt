MINIMUM FUNCTION : (Function Show Minimum Value)

Example Script 

Example 1
@echo off

:: call get_minimum array_name return_name

set /a data=%1_length - 1

for /l %%g in ( 0, 1, %data% ) do (
	if %%g equ 0 (
		set minimum=!%1[%%g]!
	)else (
		if !%1[%%g]! lss !minimum! (
			set minimum=!%1[%%g]!
		) 
	)
)

set %2=!minimum!
