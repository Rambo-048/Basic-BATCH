RANGE NUMBERS

EXAMPLE SCRIPT

--------------------------------------------------------------------------------------------------------------------------------
Example 1

@echo off
setlocal enabledelayedexpansion
goto :main

:main
setlocal

	for /l %%g in ( 5, 1, 10 ) do (
		echo %%g
	)	

endlocal
goto :eof
