Example 1

@echo off
goto :main

:: call string_find "!string!" "!find!" return char

:check_position
	if "!content:~%~1,1!" equ "!find:~%~2,1!" (
		if %~2 equ !find_limit! (
			set /a char=%~1 - %~2
			set %~3=!char!

			set found=true
		)
	)
goto :eof

:main

	call create_string content %1
	call create_string find %2
	set found=	

	set /a %3=-1

	set start_position=%4	

	if "%start_position%" equ "" (
		set start_position=0
	)

	set /a content_limit=%content_length% - 1
	set /a find_limit=%find_length% - 1

	for /l %%g in ( %start_position%, 1, %content_limit% ) do (
		set character=!content:~%%g,1!

		if "!character!" equ "!find:~0,1!" (
			for /l %%h in ( 0, 1, %find_limit% ) do (
				set /a offset=%%g + %%h

				call :check_position !offset! %%h %3
				
				if defined found (
					goto :eof
				) 				
			)
		)
	) 

goto :eof
